{{- /* Footnote written in place: text{{< footnote >}}the note, **Markdown** ok{{< /footnote >}}. Same markup as Goldmark's [^1] footnotes, so notes.js and the .footnotes styles apply (ids are fn-N/fnref-N, Goldmark's are fn:N/fnref:N). Numbered in order of appearance; partials/footnotes.html lists them at the end of the post and warns if a post also uses [^1] (separate numbering). Keyed by position so rendering the content twice doesn't number a note twice. */ -}} {{- $pos := printf "%v" .Position -}} {{- $n := index (.Page.Store.Get "footnoteNum" | default dict) $pos -}} {{- if not $n -}} {{- .Page.Store.Add "footnotes" (slice (.InnerDeindent | strings.TrimSpace | .Page.RenderString)) -}} {{- $n = len (.Page.Store.Get "footnotes") -}} {{- .Page.Store.SetInMap "footnoteNum" $pos $n -}} {{- end -}} {{ $n }} {{- /**/ -}}
