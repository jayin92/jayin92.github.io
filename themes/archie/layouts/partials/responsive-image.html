{{- /* Responsive image. Args: src, alt, title (all optional except src). /image/* paths are resolved through assets/image (mounted from content/image in config.toml) and emitted as resized WebP with srcset. Anything else falls back to a plain lazy . */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $res := "" -}} {{- if hasPrefix $src "/image/" -}} {{- $res = resources.Get (strings.TrimPrefix "/" $src) -}} {{- end -}} {{- if and $res (eq $res.ResourceType "image") (in (slice "image/png" "image/jpeg" "image/webp") $res.MediaType.Type) -}} {{- /* Resize/convert drops EXIF, so bake the rotation into the pixels first (phone photos are stored sideways). */ -}} {{- if eq $res.MediaType.Type "image/jpeg" }}{{ $res = $res.Filter images.AutoOrient }}{{ end -}} {{- $opts := "q80 photo" -}} {{- if eq $res.MediaType.Type "image/png" }}{{ $opts = "q90 drawing" }}{{ end -}} {{- $cands := slice -}} {{- range slice 640 960 1600 -}} {{- if lt . $res.Width -}} {{- $cands = $cands | append ($res.Resize (printf "%dx webp %s" . $opts)) -}} {{- end -}} {{- end -}} {{- if le $res.Width 1600 -}} {{- $cands = $cands | append ($res.Process (printf "webp %s" $opts)) -}} {{- end -}} {{- $largest := index $cands (sub (len $cands) 1) -}} {{- else -}} {{- end -}}
