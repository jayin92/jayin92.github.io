Entry fields: author (string or list), title, venue, year, url, doi, note — or `text` (Markdown) to write the entry by hand. */ -}} {{- with .Store.Get "citeKeys" -}} {{- $db := partial "reference-db.html" $ -}}

{{ i18n "references" }}

    {{- range $i, $k := . }} {{- $r := index $db $k }} {{- $n := add $i 1 }}
  1. [{{ $n }}] {{- with $r.text }}{{ . | $.RenderString }}{{ else }} {{- $authors := $r.author }} {{- if reflect.IsSlice $authors }} {{- if gt (len $authors) 5 }}{{ $authors = printf "%s, et al." (delimit (first 3 $authors) ", ") }} {{- else }}{{ $authors = delimit $authors ", " (cond (eq (len $authors) 2) " and " ", and ") }}{{ end }} {{- end }} {{- $url := $r.url | default (cond (isset $r "doi") (printf "https://doi.org/%v" $r.doi) "") }} {{- $parts := slice }} {{- with $authors }}{{ $parts = $parts | append (cond (hasSuffix . ".") . (printf "%s." .)) }}{{ end }} {{- with $r.title }} {{- $t := . | $.RenderString }} {{- $parts = $parts | append (printf "%s." (cond (ne $url "") (printf `%s` $url $t) $t)) }} {{- end }} {{- with $r.venue }}{{ $parts = $parts | append (printf "%s%s" ($.RenderString .) (cond (isset $r "year") "," ".")) }}{{ end }} {{- with $r.year }}{{ $parts = $parts | append (printf "%v." .) }}{{ end }} {{- with $r.note }}{{ $parts = $parts | append ($.RenderString .) }}{{ end }} {{- delimit $parts " " | safeHTML }} {{- end }} ↩︎
    2. {{- end }}
